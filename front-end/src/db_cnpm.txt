-- Tạo database
CREATE DATABASE IF NOT EXISTS school_bus_management;
USE school_bus_management;

-- ***************************************************************
-- 1. BẢNG QUẢN LÝ TÀI KHOẢN VÀ NGƯỜI DÙNG
-- ***************************************************************

-- Bảng: taiKhoan (Hỗ trợ đăng nhập và phân quyền)
CREATE TABLE taiKhoan (
    maTaiKhoan INT PRIMARY KEY AUTO_INCREMENT,
    tenDangNhap VARCHAR(255) NOT NULL UNIQUE,
    matKhau VARCHAR(255) NOT NULL,
    ngayTao DATETIME DEFAULT CURRENT_TIMESTAMP,
    capDo VARCHAR(50), -- Vai trò (Parent, Driver, Manager)
    trangThai VARCHAR(50)
);

-- Bảng: quanLyXe (Quản lý xe buýt, có thể Gửi tin nhắn thông báo)
CREATE TABLE quanLyXe (
    maQuanLyXe INT PRIMARY KEY AUTO_INCREMENT,
    tenQuanLyXe VARCHAR(255),
    soDienThoai VARCHAR(15),
    ngaySinh DATE,
    trangThai VARCHAR(50),
    maTaiKhoan INT UNIQUE,
    FOREIGN KEY (maTaiKhoan) REFERENCES taiKhoan(maTaiKhoan)
);

-- Bảng: taiXe (Tài xế, có thể Gửi cảnh báo)
CREATE TABLE taiXe (
    maTaiXe INT PRIMARY KEY AUTO_INCREMENT,
    tenTaiXe VARCHAR(255),
    ngaySinh DATE,
    anhTaiXe VARCHAR(255),
    soDienThoai VARCHAR(15),
    soBangLai VARCHAR(50) UNIQUE,
    trangThai VARCHAR(50),
    maTaiKhoan INT UNIQUE,
    FOREIGN KEY (maTaiKhoan) REFERENCES taiKhoan(maTaiKhoan)
);

-- Bảng: hocSinh (Học sinh)
CREATE TABLE hocSinh (
    maHocSinh INT PRIMARY KEY AUTO_INCREMENT,
    tenHocSinh VARCHAR(255),
    anhHocSinh VARCHAR(255),
    lop VARCHAR(50),
    maTaiKhoan INT, -- Nếu học sinh có tài khoản riêng
    trangThai VARCHAR(50),
    FOREIGN KEY (maTaiKhoan) REFERENCES taiKhoan(maTaiKhoan)
);

-- Bảng: phuHuynh (Phụ huynh, có thể Theo dõi vị trí và Nhận cảnh báo)
CREATE TABLE phuHuynh (
    maPhuHuynh INT PRIMARY KEY AUTO_INCREMENT, -- Đã sửa: Khóa chính được định nghĩa rõ ràng
    tenPhuHuynh VARCHAR(255),
    soDienThoai VARCHAR(15),
    ngaySinh DATE,
    maTaiKhoan INT UNIQUE,
    maHocSinh INT, -- Liên kết 1:N nếu một phụ huynh quản lý nhiều học sinh
    trangThai VARCHAR(50),
    FOREIGN KEY (maTaiKhoan) REFERENCES taiKhoan(maTaiKhoan),
    FOREIGN KEY (maHocSinh) REFERENCES hocSinh(maHocSinh)
);


-- ***************************************************************
-- 2. BẢNG QUẢN LÝ PHƯƠNG TIỆN VÀ TUYẾN ĐƯỜNG
-- ***************************************************************

-- Bảng: xeBuyt (Xe buýt)
CREATE TABLE xeBuyt (
    maXeBuyt INT PRIMARY KEY AUTO_INCREMENT,
    bienSoXe VARCHAR(50) UNIQUE,
    sucChua INT,
    mauXe VARCHAR(50),
    trangThai VARCHAR(50)
);

-- Bảng: tuyenDuong (Tuyến đường)
CREATE TABLE tuyenDuong (
    maTuyenDuong INT PRIMARY KEY AUTO_INCREMENT, -- Đã sửa: Khóa chính được định nghĩa rõ ràng
    tenTuyenDuong VARCHAR(255),
    loai VARCHAR(50)
);

-- Bảng: diemDung (Điểm dừng)
CREATE TABLE diemDung (
    maDiemDung INT PRIMARY KEY AUTO_INCREMENT,
    tenDiemDung VARCHAR(255),
    maTuyenDuong INT,
    FOREIGN KEY (maTuyenDuong) REFERENCES tuyenDuong(maTuyenDuong)
);


-- ***************************************************************
-- 3. BẢNG LỊCH TRÌNH VÀ CHUYẾN XE (Hỗ trợ Quản lý lịch trình & Phân công)
-- ***************************************************************

-- Bảng: lichTrinh (Lịch trình)
CREATE TABLE lichTrinh (
    maLichTrinh INT PRIMARY KEY AUTO_INCREMENT,
    ngay DATE,
    thoiGianDi TIME, -- Đã sửa: Sử dụng TIME thay cho nvarchar
    thoiGianDen TIME -- Đã sửa: Sử dụng TIME thay cho nvarchar
);

-- Bảng: chuyenXe (Chuyến xe - Phân công tài xế, xe, tuyến đường)
CREATE TABLE chuyenXe (
    maChuyenXe INT PRIMARY KEY AUTO_INCREMENT,
    maXeBuyt INT NOT NULL,
    maTaiXe INT NOT NULL,
    maLichTrinh INT NOT NULL,
    maTuyenDuong INT NOT NULL,
    trangThai VARCHAR(50),
    FOREIGN KEY (maXeBuyt) REFERENCES xeBuyt(maXeBuyt),
    FOREIGN KEY (maTaiXe) REFERENCES taiXe(maTaiXe),
    FOREIGN KEY (maLichTrinh) REFERENCES lichTrinh(maLichTrinh),
    FOREIGN KEY (maTuyenDuong) REFERENCES tuyenDuong(maTuyenDuong)
);

-- Bảng: phanBoHocSinh (Phân bổ học sinh vào chuyến xe & Báo cáo trạng thái)
CREATE TABLE phanBoHocSinh (
    maPhanBoHocSinh INT PRIMARY KEY AUTO_INCREMENT,
    maChuyenXe INT NOT NULL,
    maHocSinh INT NOT NULL,
    trangThai VARCHAR(50), -- Ví dụ: Assigned, Picked Up, Dropped Off
    FOREIGN KEY (maChuyenXe) REFERENCES chuyenXe(maChuyenXe),
    FOREIGN KEY (maHocSinh) REFERENCES hocSinh(maHocSinh)
);


-- ***************************************************************
-- 4. BẢNG THEO DÕI VỊ TRÍ THỜI GIAN THỰC (Hỗ trợ Giám sát thời gian thực)
-- ***************************************************************

-- Bảng: viTriThuc (Lưu trữ tọa độ GPS)
CREATE TABLE viTriThuc (
    maViTriThuc INT PRIMARY KEY AUTO_INCREMENT,
    kinhDo DECIMAL(10, 8), -- Dùng DECIMAL cho độ chính xác của tọa độ
    viDo DECIMAL(10, 8) -- Dùng DECIMAL cho độ chính xác của tọa độ
);

-- Bảng: viTriChuyenXe (Liên kết chuyến xe với vị trí và thời gian)
CREATE TABLE viTriChuyenXe (
    maViTriChuyenXe INT PRIMARY KEY AUTO_INCREMENT,
    maChuyenXe INT NOT NULL,
    maViTriThuc INT NOT NULL,
    thoiGianGhiNhan TIMESTAMP DEFAULT CURRENT_TIMESTAMP, -- Thêm trường thời gian ghi nhận
    FOREIGN KEY (maChuyenXe) REFERENCES chuyenXe(maChuyenXe),
    FOREIGN KEY (maViTriThuc) REFERENCES viTriThuc(maViTriThuc)
);

-- ***************************************************************
-- 5. BẢNG THÔNG BÁO VÀ CẢNH BÁO
-- ***************************************************************

-- Bảng: canhBao (Cảnh báo được gửi bởi Tài xế)
CREATE TABLE canhBao (
    maCanhBao INT PRIMARY KEY AUTO_INCREMENT,
    maTaiXe INT,
    noiDung TEXT, -- Dùng TEXT cho nội dung dài
    thoiGianTao DATETIME DEFAULT CURRENT_TIMESTAMP, -- Đã sửa: Sử dụng DATETIME thay cho thoigian
    FOREIGN KEY (maTaiXe) REFERENCES taiXe(maTaiXe)
);

-- Bảng: chiTietCanhBao (Chi tiết người nhận Cảnh báo - Phụ huynh)
CREATE TABLE chiTietCanhBao (
    maChiTietCanhBao INT PRIMARY KEY AUTO_INCREMENT,
    maCanhBao INT NOT NULL,
    maPhuHuynh INT NOT NULL,
    FOREIGN KEY (maCanhBao) REFERENCES canhBao(maCanhBao),
    FOREIGN KEY (maPhuHuynh) REFERENCES phuHuynh(maPhuHuynh)
);

-- Bảng: thongBao (Thông báo được gửi bởi Quản lý xe)
CREATE TABLE thongBao (
    maThongBao INT PRIMARY KEY AUTO_INCREMENT,
    maQuanLyXe INT,
    noiDung TEXT, -- Dùng TEXT cho nội dung dài
    thoiGianTao DATETIME DEFAULT CURRENT_TIMESTAMP, -- Đã sửa: Sử dụng DATETIME thay cho thoigian
    FOREIGN KEY (maQuanLyXe) REFERENCES quanLyXe(maQuanLyXe)
);

-- Bảng: chiTietThongBao (Chi tiết người nhận Thông báo)
CREATE TABLE chiTietThongBao (
    maChiTietThongBao INT PRIMARY KEY AUTO_INCREMENT,
    maThongBao INT NOT NULL,
    maTaiKhoan INT NOT NULL, -- Người nhận có thể là Tài xế, Phụ huynh hoặc Quản lý khác
    FOREIGN KEY (maThongBao) REFERENCES thongBao(maThongBao),
    FOREIGN KEY (maTaiKhoan) REFERENCES taiKhoan(maTaiKhoan)
);
